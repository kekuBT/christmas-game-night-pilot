// Scratch reproduction (not a gate driver): a guest still on a casual Math Mayhem game's results when the host makes the
// room a Family Cup. Does that results screen keep calling the game Casual, or does it turn into "Practice"?
// Usage: node repro-results-after-cup.mjs <baseUrl> <worktree>
const BASE = process.argv[2];
const WT = process.argv[3];
const { launchChromium } = await import(`${WT}/scripts/lib/driver-env.mjs`);
const { PROTOCOL_HEADERS } = await import(`${WT}/scripts/lib/protocol.mjs`);
const { solveMath } = await import(`${WT}/scripts/lib/math-solve.mjs`);
const sleep = (ms) => new Promise((r) => setTimeout(r, ms));
async function api(path, { token, body } = {}) {
  const headers = { "content-type": "application/json", ...PROTOCOL_HEADERS };
  if (token) headers["x-session-token"] = token;
  const r = await fetch(`${BASE}${path}`, { method: body ? "POST" : "GET", headers, body: body ? JSON.stringify(body) : undefined });
  const j = await r.json().catch(() => ({}));
  if (r.status >= 400) throw new Error(`${path}: ${r.status} ${JSON.stringify(j)}`);
  return j;
}
const intent = (roomId, seat, gid, action, payload) =>
  api(`/rooms/${roomId}/intents`, { token: seat.token, body: { intent: { kind: "intent", schemaVersion: 1, clientEventId: crypto.randomUUID(), roomId, gameInstanceId: gid, action, payload } } });

const room = await api("/rooms", { body: { name: "Repro", kind: "casual" } });
const h = await api(`/rooms/${room.roomId}/join`, { body: { displayName: "Keku", asHost: true, hostClaimToken: room.hostClaimToken } });
const g = await api("/rooms/join-by-code", { body: { joinCode: room.joinCode, displayName: "Amina" } });
const seats = [{ playerId: h.playerId, token: h.recoveryToken }, { playerId: g.playerId, token: g.recoveryToken }];
console.log("room kind at start:", (await api(`/rooms/${room.roomId}/projection?audience=tv`)).roomKind);

const browser = await launchChromium();
let failures = 0;
try {
  const ctx = await browser.newContext({ viewport: { width: 390, height: 844 } });
  const amina = await ctx.newPage();
  await amina.goto(`${BASE}/join`, { waitUntil: "domcontentloaded" });
  await amina.evaluate((s) => { sessionStorage.setItem("cgn_roomId", s.roomId); sessionStorage.setItem("cgn_playerId", s.playerId); sessionStorage.setItem("cgn_token", s.token); }, { ...seats[1], roomId: room.roomId });
  await amina.goto(`${BASE}/room`, { waitUntil: "domcontentloaded" });
  await sleep(1500);

  const { gameId: gid } = await api(`/rooms/${room.roomId}/games`, { token: seats[0].token, body: { gameId: "math-mayhem", official: false } });
  for (const s of seats) await intent(room.roomId, s, gid, "briefing-ready", { ready: true });
  for (let t = 0; t < 50; t++) { if ((await api(`/rooms/${room.roomId}/projection?audience=tv`)).games.find((x) => x.id === gid)?.status === "playing") break; await sleep(200); }
  for (const [i, s] of seats.entries()) {
    const mine = (await api(`/rooms/${room.roomId}/projection?audience=player`, { token: s.token })).myGames.find((x) => x.id === gid);
    for (const [n, q] of mine.questions.entries()) {
      const o = await intent(room.roomId, s, gid, "math-open", { questionId: q.id });
      const idx = o.result.options.findIndex((x) => (n < (i === 0 ? 5 : 2) ? Number(x) === solveMath(o.result.prompt) : Number(x) !== solveMath(o.result.prompt)));
      await intent(room.roomId, s, gid, "submit-math-answer", { questionId: q.id, chosenIndex: idx });
    }
  }
  await sleep(3000);
  const before = (await amina.locator("body").innerText()).replace(/\s+/g, " ");
  console.log("Amina's results BEFORE the change:", before.slice(0, 400));
  const beforeLegacy = await amina.getByTestId("show-results").count();

  await api(`/rooms/${room.roomId}/cup`, { token: seats[0].token, body: {} });
  console.log("room kind after:", (await api(`/rooms/${room.roomId}/projection?audience=tv`)).roomKind);
  await sleep(3000);
  const after = (await amina.locator("body").innerText()).replace(/\s+/g, " ");
  console.log("Amina's results AFTER the change:", after.slice(0, 600));
  const afterLegacy = await amina.getByTestId("show-results").count();
  await amina.screenshot({ path: `${process.env.OUT ?? "."}/repro-results-after-cup-390.png`, fullPage: true });
  await amina.reload({ waitUntil: "domcontentloaded" });
  await sleep(3000);
  const reloaded = (await amina.locator("body").innerText()).replace(/\s+/g, " ");
  console.log("Amina's results AFTER a reload:", reloaded.slice(0, 600));
  const says = (t) => ({ practice: /Practice/i.test(t), casual: /Casual/i.test(t) });
  console.log("legacy results card (show-results) before/after:", beforeLegacy, afterLegacy);
  for (const [label, t] of [["before", before], ["after", after], ["after reload", reloaded]]) {
    const s = says(t);
    const ok = s.casual && !s.practice;
    if (!ok) failures++;
    console.log(`${ok ? "PASS" : "FAIL"} ${label}: the casual game's results say Casual, never Practice — ${JSON.stringify(s)}`);
  }
} finally { await browser.close(); }
console.log(failures === 0 ? "REPRO: no fault" : `REPRO: ${failures} FAILED`);
process.exit(failures === 0 ? 0 : 1);
