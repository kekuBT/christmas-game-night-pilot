// The inputs for the zod exposure assessment: for each network-facing schema with a collection, the worst bodies
// that fit the server's 64 KiB cap (MAX_MESSAGE_BYTES; a body of exactly 65,536 bytes is accepted, one more is refused).
export const CAP = 64 * 1024;

/** `prefix` + as many `unit`s as fit, comma-separated + `suffix`, at most CAP bytes. */
export function fill(prefix, unit, suffix, cap = CAP) {
  const room = cap - Buffer.byteLength(prefix) - Buffer.byteLength(suffix);
  const per = Buffer.byteLength(unit) + 1;
  const n = Math.max(0, Math.floor((room + 1) / per));
  const body = prefix + Array(n).fill(unit).join(",") + suffix;
  if (Buffer.byteLength(body) > cap) throw new Error("over the cap");
  return { body, elements: n };
}
const deep = (prefix, suffix, cap = CAP) => {
  const d = Math.floor((cap - Buffer.byteLength(prefix) - Buffer.byteLength(suffix)) / 2);
  return { body: prefix + "[".repeat(d) + "]".repeat(d) + suffix, elements: 1, depth: d };
};

/** Make "k" keys unique: "k0","k1",... while staying under the cap. */
export function buildBody(id) {
  const v = VARIANTS[id];
  if (!v.keysUnique) return v.build();
  const { body } = v.build();
  const head = body.slice(0, body.indexOf('"k":0'));
  const tail = body.slice(body.lastIndexOf('"k":0') + 5);
  const parts = [];
  let len = Buffer.byteLength(head) + Buffer.byteLength(tail);
  for (let i = 0; ; i++) {
    const p = `"k${i}":0`;
    const add = Buffer.byteLength(p) + (parts.length ? 1 : 0);
    if (len + add > CAP) break;
    parts.push(p); len += add;
  }
  return { body: head + parts.join(",") + tail, elements: parts.length };
}
