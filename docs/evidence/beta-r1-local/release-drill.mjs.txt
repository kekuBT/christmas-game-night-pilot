// Beta R1 release drill (docs/beta/beta-r1-release.md), on this machine with real packages:
// pack PREV and NEW from clean checkouts, install each with --omit=dev, lay them out as releases/<sha> with a
// `current` link, then: run PREV on a data dir outside both; stop; back up (NEW's tool); switch to NEW; check;
// play into it; stop; roll back to PREV; check it reads NEW's data; then restore the pre-deploy backup into a new
// folder and serve it. Usage: node release-drill.mjs <prev-checkout> <new-checkout>
import { execFileSync, spawn } from "node:child_process";
import { existsSync, mkdirSync, mkdtempSync, readFileSync, realpathSync, rmSync, symlinkSync, unlinkSync } from "node:fs";
import { tmpdir } from "node:os";
import { join } from "node:path";
const [prevTree, newTree] = process.argv.slice(2);
let failures = 0;
const check = (name, ok, detail = "") => { console.log(`${ok ? "PASS" : "FAIL"} ${name}${detail ? ` — ${detail}` : ""}`); if (!ok) failures++; };
const T = mkdtempSync(join(tmpdir(), "mz-release-drill-"));
const procs = [];
const CODE = "release-drill-code-1234", ORIGIN = "https://beta.example.test";
const data = join(T, "var-lib-cgn");
const opt = join(T, "opt-cgn");
const current = join(opt, "current");
const env = { NODE_ENV: "production", MCHEZO_DATA_DIR: data, MCHEZO_PUBLIC_ORIGIN: ORIGIN, MCHEZO_BETA_ACCESS_CODE: CODE, MCHEZO_TRUST_PROXY: "1", MCHEZO_LISTEN_HOST: "127.0.0.1" };
function pack(tree, label) {
  const sha = execFileSync("git", ["rev-parse", "HEAD"], { cwd: tree, encoding: "utf8" }).trim();
  const dirty = execFileSync("git", ["status", "--porcelain"], { cwd: tree, encoding: "utf8" }).trim();
  const out = join(T, `pkg-${label}`);
  execFileSync(process.execPath, ["scripts/release/pack.mjs", "--out", out], { cwd: tree, stdio: ["ignore", "ignore", "inherit"] });
  execFileSync("tar", ["-czf", `${out}.tgz`, "-C", out, "."]);
  const hash = execFileSync("sha256sum", [`${out}.tgz`], { encoding: "utf8" }).split(" ")[0];
  const dest = join(opt, "releases", sha);
  mkdirSync(dest, { recursive: true });
  execFileSync("tar", ["-xzf", `${out}.tgz`, "-C", dest]);
  execFileSync("npm", ["ci", "--omit=dev", "--ignore-scripts", "--no-audit", "--no-fund", "--prefer-offline"], { cwd: dest, stdio: "ignore" });
  const rev = JSON.parse(readFileSync(join(dest, "release.json"), "utf8")).revision;
  check(`${label}: packed from a clean checkout of ${sha.slice(0, 7)}, tarball sha256 ${hash.slice(0, 16)}…, release.json revision matches`, !dirty && rev === sha, `revision ${rev}`);
  return { sha, dest };
}
const link = (dest) => { if (existsSync(current)) unlinkSync(current); symlinkSync(dest, current); };
function start(extra = {}, label) {
  const child = spawn(process.execPath, ["dist/server/serve.js"], { cwd: current, env: { PATH: process.env.PATH, HOME: process.env.HOME, PORT: "0", ...env, ...extra }, stdio: ["ignore", "pipe", "pipe"] });
  const p = { label, child, out: "" }; procs.push(p);
  child.stdout.on("data", (d) => { p.out += d; }); child.stderr.on("data", (d) => { p.out += d; });
  return new Promise((resolve) => {
    const t = setInterval(() => { const m = /on http:\/\/localhost:(\d+)/.exec(p.out); if (m) { clearInterval(t); resolve({ port: Number(m[1]), p }); } }, 100);
    child.once("exit", (code) => { clearInterval(t); resolve({ exit: code, p }); });
  });
}
const stop = (p) => new Promise((r) => { if (p.child.exitCode !== null) return r(p.child.exitCode); p.child.once("exit", (c) => r(c)); p.child.kill("SIGTERM"); });
async function session(port) {
  const base = `http://127.0.0.1:${port}`;
  const enter = await fetch(`${base}/beta/enter`, { method: "POST", redirect: "manual", headers: { "content-type": "application/x-www-form-urlencoded" }, body: `code=${CODE}&next=/` });
  const cookie = (enter.headers.get("set-cookie") ?? "").split(";")[0];
  const api = (path, body) => fetch(`${base}${path}`, { method: body ? "POST" : "GET", headers: { cookie, "content-type": "application/json" }, body: body ? JSON.stringify(body) : undefined }).then(async (r) => ({ status: r.status, body: await r.json().catch(() => ({})) }));
  const health = await (await fetch(`${base}/health`)).json();
  return { api, health };
}
const names = async (s, roomId) => { const r = await s.api(`/rooms/${roomId}/projection?audience=tv`); return r.status === 200 ? (r.body.players ?? []).map((x) => x.displayName).sort().join(",") : `HTTP ${r.status}`; };
try {
  const prev = pack(prevTree, "PREV");
  const next = pack(newTree, "NEW");
  // PREV is live; the family plays into it. The operator creates the data directory (0700) as the architecture says;
  // 8dfe323 (PREV) would crash on a missing one (B-08, fixed after it).
  mkdirSync(data, { recursive: true, mode: 0o700 });
  link(prev.dest);
  let s = await start({}, "prev-1");
  let c = await session(s.port);
  check("PREV runs from releases/<sha> through `current`, data outside it", c.health.revision === prev.sha, JSON.stringify(c.health));
  const room = await c.api("/rooms", { name: "Drill night", kind: "casual" });
  await c.api("/rooms/join-by-code", { joinCode: room.body.joinCode, displayName: "Keku", asHost: true, hostClaimToken: room.body.hostClaimToken });
  await c.api("/rooms/join-by-code", { joinCode: room.body.joinCode, displayName: "Amina" });
  const roomId = room.body.roomId;
  check("PREV: a room with two players", (await names(c, roomId)) === "Amina,Keku", await names(c, roomId));
  // Deploy NEW, between sessions.
  check("deploy 1/4: stop PREV (SIGTERM, exit 0, lock freed)", (await stop(s.p)) === 0 && !existsSync(join(data, "server.lock")));
  const backupDir = join(T, "var-backups", "before-new");
  const tool = (...a) => { try { return { code: 0, out: execFileSync(process.execPath, ["dist/server/datatool.js", ...a], { cwd: next.dest, encoding: "utf8" }) }; } catch (e) { return { code: e.status, out: `${e.stdout ?? ""}${e.stderr ?? ""}` }; } };
  const b = tool("backup", "--data", data, "--out", backupDir);
  check("deploy 2/4: back up with the server stopped (NEW's tool)", b.code === 0 && /^backup OK: 1 room\(s\)/.test(b.out), b.out.trim());
  link(next.dest);
  s = await start({}, "new-1");
  c = await session(s.port);
  check("deploy 3/4: switch `current` to NEW and start: /health names NEW", c.health.revision === next.sha, JSON.stringify(c.health));
  check("deploy 4/4: the restore line reports the room, nothing not loaded", /\[restore\] 1 room\(s\) restored, 0 not loaded/.test(s.p.out) && (await names(c, roomId)) === "Amina,Keku", (s.p.out.match(/\[restore\][^\n]*/) ?? [""])[0]);
  const joined = await c.api("/rooms/join-by-code", { joinCode: room.body.joinCode, displayName: "Baraka" });
  check("NEW: play goes on (a third player joins)", joined.status === 201 && (await names(c, roomId)) === "Amina,Baraka,Keku");
  // Roll back.
  check("rollback 1/3: stop NEW (exit 0)", (await stop(s.p)) === 0);
  link(prev.dest);
  s = await start({}, "prev-2");
  c = await session(s.port);
  check("rollback 2/3: `current` back to PREV: /health names PREV", c.health.revision === prev.sha, JSON.stringify(c.health));
  check("rollback 3/3: PREV reads what NEW wrote (three players), nothing quarantined", (await names(c, roomId)) === "Amina,Baraka,Keku" && /\[restore\] 1 room\(s\) restored, 0 not loaded/.test(s.p.out) && !/NOT LOADED|to .*quarantine/.test(s.p.out), `${await names(c, roomId)} | ${(s.p.out.match(/\[restore\][^\n]*/) ?? [""])[0]}`);
  await stop(s.p);
  // Restore the pre-deploy backup into a new folder and serve it with NEW.
  link(next.dest);
  const restored = join(T, "restore-drill");
  const r = tool("restore", "--backup", backupDir, "--into", restored);
  s = await start({ MCHEZO_DATA_DIR: restored }, "new-restore");
  c = await session(s.port);
  check("restore drill: the pre-deploy backup, restored into a new folder, serves the room as it was (two players)", r.code === 0 && (await names(c, roomId)) === "Amina,Keku", `${r.out.trim()} | ${await names(c, roomId)}`);
  await stop(s.p);
  check("the live data was never touched by the restore (still three players on disk)", JSON.parse(readFileSync(join(data, `${roomId}.json`), "utf8")).room.players.length === 3);
} catch (e) {
  check("the drill ran to the end", false, String(e?.stack ?? e).split("\n").slice(0, 3).join(" | "));
} finally {
  if (failures) for (const p of procs) console.log(`--- ${p.label} ---\n${p.out.slice(-1200)}`);
  for (const p of procs) if (p.child.exitCode === null) p.child.kill("SIGKILL");
  rmSync(T, { recursive: true, force: true });
  console.log(`cleanup: ${procs.length} server(s) stopped; ${existsSync(T) ? "left on disk: " + T : "nothing left on disk"}`);
}
console.log(failures ? `RELEASE DRILL FAILED: ${failures}` : "RELEASE DRILL PASSED");
process.exit(failures ? 1 : 0);
